```javascript
class ThemePanel {
    constructor() {
        // 单例模式保证唯一实例
        this.panel = null;
        this.isVisible = false;
        this.currentTheme = GM_getValue('github_theme', 'default');
    }
    
    // 懒加载创建面板
    createPanel() {
        if (this.panel) return this.panel;
        // 首次调用时才创建 DOM
    }
    
    // 状态管理模式
    switchTheme(themeKey) {
        GM_setValue('github_theme', themeKey); // 持久化
        this.currentTheme = themeKey;           // 内存状态
        this.updateUI();                        // UI 同步
        applyTheme(themeKey);                   // 实际应用
    }
}
```

## 🔧 实现细节解析

### **1. 样式注入的精确控制**

```javascript
// 问题：GitHub 的动态加载会覆盖自定义样式
// 解决：多层次样式注入 + 优先级控制

function applyTheme(themeKey) {
    // 1. 清除旧样式，避免冲突
    const oldStyle = document.getElementById('github-theme-custom-style');
    if (oldStyle) oldStyle.remove();
    
    // 2. 默认主题特殊处理 - 完全清除自定义
    if (themeKey === 'default') return;
    
    // 3. 双重注入保证生效
    const styleEl = document.createElement('style');
    styleEl.id = 'github-theme-custom-style';
    styleEl.textContent = generateThemeCSS(theme);
    document.head.appendChild(styleEl);
}
```

### **2. 搜索过滤的防抖优化**

```javascript
setupSearch() {
    const searchInput = document.getElementById('theme-search');
    let debounceTimer;
    
    searchInput.addEventListener('input', (e) => {
        // 清除上一次的定时器
        clearTimeout(debounceTimer);
        
        // 200ms 后执行搜索，减少 DOM 操作频率
        debounceTimer = setTimeout(() => {
            this.searchQuery = e.target.value;
            this.renderThemeButtons(this.searchQuery);
        }, 200);
    });
}
```

### **3. 事件系统的分层处理**

```javascript
// 多层事件监听，处理不同的关闭场景
setupEventListeners() {
    // 1. 按钮关闭
    closeBtn.addEventListener('click', () => this.hide());
    
    // 2. 外部点击关闭（冒泡处理）
    document.addEventListener('click', (e) => {
        if (this.isVisible && !this.panel.contains(e.target)) {
            this.hide();
        }
    });
    
    // 3. 键盘快捷键
    document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && this.isVisible) {
            this.hide();
        }
    });
}
```
